from __future__ import annotations

import sqlite3
import sys
import time

DB = sys.argv[1]
ACTION = sys.argv[2]  # plan | time | analyze
FLOOR = 1024 * 1024  # dedup default floor placeholder; overridden by argv[3]
if len(sys.argv) > 3:
    FLOOR = int(sys.argv[3])

DISTINCT = """
    SELECT size FROM files
    WHERE is_dir = 0 AND size > 0 AND is_cloud_placeholder = 0
    GROUP BY size, (ino = 0), CASE WHEN ino = 0 THEN 0 ELSE dev END,
             CASE WHEN ino = 0 THEN rowid ELSE ino END
"""
QUAL = f"SELECT size FROM ({DISTINCT}) GROUP BY size HAVING COUNT(*) >= 2 AND (COUNT(*) - 1) * size >= ?"
DISTINCT_NI = DISTINCT.replace("FROM files", "FROM files NOT INDEXED")
QUAL_NI = f"SELECT size FROM ({DISTINCT_NI}) GROUP BY size HAVING COUNT(*) >= 2 AND (COUNT(*) - 1) * size >= ?"
QUERIES = {
    "v1_count": f"SELECT COUNT(*) AS total FROM files WHERE is_dir = 0 AND size > 0 AND is_cloud_placeholder = 0 AND size IN ({QUAL_NI})",
    "v1_stream": f"SELECT * FROM files WHERE is_dir = 0 AND size > 0 AND is_cloud_placeholder = 0 AND size IN ({QUAL_NI}) ORDER BY size",
    "v1_immaterial": f"SELECT COUNT(*) AS bucket_count, COALESCE(SUM((c - 1) * size), 0) FROM (SELECT size, COUNT(*) AS c FROM ({DISTINCT_NI}) GROUP BY size HAVING COUNT(*) >= 2 AND (COUNT(*) - 1) * size < ?)",
    "v2_count": f"SELECT COUNT(*) AS total FROM files NOT INDEXED WHERE is_dir = 0 AND size > 0 AND is_cloud_placeholder = 0 AND size IN ({QUAL_NI})",
    "count": f"SELECT COUNT(*) AS total FROM files WHERE is_dir = 0 AND size > 0 AND is_cloud_placeholder = 0 AND size IN ({QUAL})",
    "stream": f"SELECT * FROM files WHERE is_dir = 0 AND size > 0 AND is_cloud_placeholder = 0 AND size IN ({QUAL}) ORDER BY size",
    "immaterial": f"SELECT COUNT(*) AS bucket_count, COALESCE(SUM((c - 1) * size), 0) FROM (SELECT size, COUNT(*) AS c FROM ({DISTINCT}) GROUP BY size HAVING COUNT(*) >= 2 AND (COUNT(*) - 1) * size < ?)",
    "qualifying": QUAL,
    "old_count": "SELECT COUNT(*) AS total FROM files WHERE is_dir = 0 AND size > 0 AND is_cloud_placeholder = 0 AND size IN (SELECT size FROM files WHERE is_dir = 0 AND size > 0 AND is_cloud_placeholder = 0 GROUP BY size HAVING COUNT(*) >= 2 AND (COUNT(*) - 1) * size >= ?)",
}

con = sqlite3.connect(DB)
print("sqlite", sqlite3.sqlite_version, "stat1 rows:",
      con.execute("select count(*) from sqlite_master where name='sqlite_stat1'").fetchone()[0] and
      con.execute("select count(*) from sqlite_stat1").fetchone()[0])
if ACTION == "analyze":
    t = time.perf_counter()
    con.execute("PRAGMA analysis_limit=0")
    con.execute("ANALYZE files")
    con.commit()
    print("ANALYZE s", round(time.perf_counter() - t, 1))
    print(con.execute("select * from sqlite_stat1").fetchall())
elif ACTION == "plan":
    for name, q in QUERIES.items():
        print("==", name)
        for r in con.execute("EXPLAIN QUERY PLAN " + q, (FLOOR,)):
            print("  ", r[0], r[1], r[2], r[3])
elif ACTION == "time":
    for name in sys.argv[4:] or QUERIES:
        t = time.perf_counter(); c0 = time.process_time()
        cur = con.execute(QUERIES[name], (FLOOR,))
        n = 0
        for _ in cur:
            n += 1
        print(name, "rows", n, "wall_s", round(time.perf_counter() - t, 1), "cpu_s", round(time.process_time() - c0, 1), flush=True)
